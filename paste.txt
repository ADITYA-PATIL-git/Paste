async function downloadFile(pdfbase64, pdfname) {
  try {
    setIsLoading(true);

    if (Platform.OS === 'android' && Platform.Version < 29) {
      const granted = await PermissionsAndroid.request(
        PermissionsAndroid.PERMISSIONS.WRITE_EXTERNAL_STORAGE,
      );

      if (granted !== PermissionsAndroid.RESULTS.GRANTED) {
        Alert.alert('Permission denied');
        setIsLoading(false);
        return;
      }
    }

    const { fs } = ReactNativeBlobUtil;
    const { DownloadDir } = fs.dirs;
    const filePath = `${DownloadDir}/${pdfname}.pdf`;

    // Write the file first
    await fs.writeFile(filePath, pdfbase64, 'base64');
    console.log("File saved:", filePath);

    // Now add it to the download manager
    ReactNativeBlobUtil.config({
      addAndroidDownloads: {
        useDownloadManager: true,
        notification: true,
        path: filePath,
        title: `${pdfname}.pdf`,
        mime: 'application/pdf',
        mediaScannable: true,
        description: "Downloading file...",
      }
    })
      .fetch('GET', `file://${filePath}`)
      .then(() => {
        console.log("Download Manager notified!");
        FileViewer.open(filePath);
      })
      .catch(err => {
        console.log("Error:", err);
        Alert.alert("Download failed");
      })
      .finally(() => setIsLoading(false));

  } catch (err) {
    console.log("Catch:", err);
    Alert.alert("Something went wrong");
    setIsLoading(false);
  }
}
