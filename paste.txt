setRevokeDate(prev => {
  const updated = prev.map((d, i) =>
    i === idx ? { ...d, checked: e ? 'Y' : 'N' } : d
  );

  const selected = updated
    .filter(d => d.checked === 'Y')
    .map(d => ({ Dates: d.LEAVE_DATE }));

  setRevokeSaveDate(selected);          // update saved dates
  setRevokeDataLength(selected.length); // update count
  setSelectAll(updated.every(d => d.checked === 'Y')); // sync select all state

  return updated;
});
