import React, { useEffect, useState, useContext } from 'react';
import { View, Text, Image, Dimensions, StyleSheet } from 'react-native';
import { useNavigation } from '@react-navigation/native';
import {
  createDrawerNavigator,
  DrawerContentScrollView,
  DrawerItem,
} from '@react-navigation/drawer';
import 'react-native-gesture-handler';
import { HeaderNotificationUser } from '../components/HeaderIcons';
import BottomTabNavigator from '../components/BottomTabNavigator';
import MyWorkSpace from './MyWorkSpace';
import DrawerButton from '../components/DrawerButton';
import {
  getFocusedRouteNameFromRoute,
} from '@react-navigation/native';
import DetailScreen from './DetailScreen';
import SearchScreen from './SearchScreen';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import { SearchContext } from '../context/SearchContext';
import { encryptUsingAES256 } from '../Encryption';
import { decryptUsingAES256 } from '../decryption';
import { FetchHttpRequest } from '../src/FetchHttpRequest';
import AsyncStorage from '@react-native-async-storage/async-storage';

const { width } = Dimensions.get('window');

const RootScreen = ({ route }) => {
  const {
    userData,
    companyCode,
    bday,
    birthdayData,
    anniversaryData,
    companyName,
  } = route.params;

  const Stack = createNativeStackNavigator();
  const Drawer = createDrawerNavigator();
  const navigation = useNavigation();

  const [loading, setLoading] = useState(false);
  const [profileData, setProfileData] = useState('');
  const [menuList, setMenuList] = useState([]);

  const { query, setQuery } = useContext(SearchContext);

  useEffect(() => {
    fillApplicationMenuList();
  }, []);

  useEffect(() => {
    const fetchMenuList = async () => {
      try {
        const jsonValue = await AsyncStorage.getItem('my_key');
        if (jsonValue) {
          let parsed = JSON.parse(jsonValue);
          console.log('Fetched from storage:', parsed);

          // if it's a stringified array, parse it again
          if (typeof parsed === 'string') {
            try {
              parsed = JSON.parse(parsed);
            } catch (err) {
              console.log('Second parse failed:', err);
              parsed = [];
            }
          }

          if (Array.isArray(parsed)) {
            setMenuList(parsed);
          } else if (parsed && parsed.Table && Array.isArray(parsed.Table)) {
            setMenuList(parsed.Table);
          } else {
            console.log('Unexpected format for menu list:', parsed);
            setMenuList([]);
          }
        }
      } catch (e) {
        console.log('Error reading menu list from AsyncStorage:', e);
      }
    };

    fetchMenuList();
  }, []);

  console.log('Final menuList:', Array.isArray(menuList), menuList);

  async function fillApplicationMenuList() {
    try {
      const CompanyName = await AsyncStorage.getItem('COMPNAME');
      const UserIdData = await AsyncStorage.getItem('UserId');
      const EmpRoleId = await AsyncStorage.getItem('id');

      const param = {
        sUserId: parseInt(UserIdData),
        sCompanyCode: CompanyName?.trim(),
        P_ROLE_ID: parseInt(EmpRoleId),
      };

      const encrypted = encryptUsingAES256(param).toString();

      const Config = {
        Url: 'Security/MenuMaster/FillMobileMenu',
        body: {
          sInput: encrypted,
        },
      };

      const resp = await FetchHttpRequest(Config);
      const decrypt = decryptUsingAES256(resp);
      console.log('decrypt =======', decrypt);

      let data = decrypt;
      if (typeof decrypt === 'string') {
        try {
          data = JSON.parse(decrypt);
        } catch (err) {
          console.log('Could not parse decrypt:', err);
          data = [];
        }
      }

      await AsyncStorage.setItem('my_key', JSON.stringify(data));
      setLoading(false);
    } catch (err) {
      console.log('Error in fillApplicationMenuList:', err);
    }
  }

  useEffect(() => {
    if (userData?.data?.Original_Photo) {
      EmployeeProfilePhoto();
    }
  }, [userData]);

  async function EmployeeProfilePhoto() {
    try {
      const param = {
        FileName: userData.data.sProfilePhoto,
        Identity: 'Profile',
        sCompanyCode: companyCode.trim(),
      };

      const encrypted = encryptUsingAES256(param).toString();

      const Config = {
        Url: `DownloadLogo`,
        body: {
          sInput: encrypted,
        },
        navigation,
      };

      const resp = await FetchHttpRequest(Config);
      await AsyncStorage.setItem('base64', resp?.data);

      if (resp?.data) {
        setProfileData(resp?.data);
        setLoading(false);
      } else {
        console.log('Profile image not found');
        setProfileData('');
        setLoading(false);
      }
    } catch (err) {
      console.log('Error in EmployeeProfilePhoto:', err);
    }
  }

  function CustomDrawerContent(props) {
    return (
      <DrawerContentScrollView {...props}>
        {profileData ? (
          <Image
            style={styles.image1}
            source={{ uri: `data:image/jpeg;base64,${profileData}` }}
          />
        ) : null}

        <Text style={styles.drawerHeader}>Menu</Text>

        {Array.isArray(menuList) && menuList.length > 0 ? (
          menuList.map((item, index) => (
            <DrawerItem
              key={index}
              label={item.MENU_NAME}
              onPress={() => props.navigation.navigate('HomeScreenTabs')}
            />
          ))
        ) : (
          <Text style={styles.noMenuText}>No menu items available</Text>
        )}

        <DrawerItem
          label="Home Screen"
          onPress={() => props.navigation.navigate('HomeScreenTabs')}
        />
        <DrawerItem
          label="Workspace"
          onPress={() =>
            props.navigation.navigate('Workspace', { screen: 'Workspace' })
          }
        />
      </DrawerContentScrollView>
    );
  }

  function SearchStack() {
    return (
      <Stack.Navigator
        screenOptions={{
          headerShown: false,
        }}
      >
        <Stack.Screen name="SearchScreen" component={SearchScreen} />
        <Stack.Screen name="DetailScreen" component={DetailScreen} />
      </Stack.Navigator>
    );
  }

  return (
    <Drawer.Navigator
      screenOptions={{
        drawerStatusBarAnimation: 'fade',
        drawerType: 'slide',
        overlayColor: 'transparent',
        sceneStyle: { backgroundColor: '#eef2fe' },
        headerTintColor: '#fff',
        headerShadowVisible: false,
        title: false,
        headerRight: () => <HeaderNotificationUser />,
        headerLeft: () => <DrawerButton />,
        freezeOnBlur: true,
      }}
      drawerContent={props => <CustomDrawerContent {...props} />}
    >
      <Drawer.Screen
        name="HomeScreenTabs"
        options={({ route }) => {
          const routeName = getFocusedRouteNameFromRoute(route) ?? 'Home';
          return {
            headerStyle: {
              backgroundColor: '#2860E7',
              borderBottomLeftRadius: routeName === 'Home' ? 0 : 14,
              borderBottomRightRadius: routeName === 'Home' ? 0 : 14,
            },
            headerTitle: routeName === 'Home' ? '' : routeName,
          };
        }}
      >
        {props => (
          <BottomTabNavigator
            {...props}
            userData={userData}
            profileData={profileData}
            bday={bday}
            birthdayData={birthdayData}
            anniversaryData={anniversaryData}
          />
        )}
      </Drawer.Screen>

      <Drawer.Screen
        name="Workspace"
        component={MyWorkSpace}
        options={{
          headerStyle: { backgroundColor: '#2860E7' },
        }}
      />

      <Drawer.Screen
        name="SearchStack"
        component={SearchStack}
        options={{
          headerStyle: { backgroundColor: '#2860E7' },
        }}
      />
    </Drawer.Navigator>
  );
};

const styles = StyleSheet.create({
  image1: {
    borderRadius: 100,
    height: width * 0.2,
    width: width * 0.2,
    marginLeft: width * 0.028,
  },
  drawerHeader: {
    fontSize: 16,
    fontWeight: 'bold',
    marginVertical: 10,
    marginLeft: 15,
  },
  noMenuText: {
    marginLeft: 20,
    color: 'gray',
  },
});

export default RootScreen;
