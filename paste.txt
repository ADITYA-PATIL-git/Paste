import * as React from 'react';
import {
  View,
  Text,
  Pressable,
  KeyboardAvoidingView,
  Image,
  Alert,
  ScrollView,
  Platform,
  ActivityIndicator,
} from 'react-native';
import { useNavigation } from '@react-navigation/native';
import { decryptUsingAES256, encryptUsingAES256 } from '../utils/encryption'; // adjust path
import { FetchHttpRequest } from '../utils/network'; // adjust path
import { GetBirthdays } from '../api/GetBirthdays'; // adjust path

const LoginScreen = () => {
  const navigation = useNavigation();

  const [companyCode, setCompanyCode] = React.useState('');
  const [username, setUsername] = React.useState('');
  const [password, setPassword] = React.useState('');

  const [birthdayData, setBirthdayData] = React.useState([]);
  const [userData, setUserData] = React.useState(null);
  const [readyToNavigate, setReadyToNavigate] = React.useState(false);
  const [loading, setLoading] = React.useState(false);
  const [bday, setBday] = React.useState(false);
  const [compId, setCompId] = React.useState(null);

  // 🔹 1️⃣ Top-level useEffect that triggers navigation when data is ready
  React.useEffect(() => {
    if (readyToNavigate && birthdayData.length > 0 && userData) {
      console.log('✅ Navigating with birthdayData:', birthdayData);

      navigation.reset({
        index: 0,
        routes: [
          {
            name: 'RootScreen',
            params: {
              userData: userData,
              companyCode: companyCode,
              bday: bday,
              birthdayData: birthdayData,
            },
          },
        ],
      });

      // reset so navigation doesn’t fire twice
      setReadyToNavigate(false);
      setLoading(false);
    }
  }, [readyToNavigate, birthdayData, userData]);

  // 🔹 2️⃣ Fetch Birthdays (Reusable)
  const fetchBirthdays = async compId => {
    try {
      const birthdayList = await GetBirthdays(compId);
      setBirthdayData(birthdayList || []);
    } catch (error) {
      console.log('Error fetching birthdays:', error);
      setBirthdayData([]);
    }
  };

  // 🔹 3️⃣ Login Function
  const doLogin = async () => {
    setLoading(true);
    const Appversion = '0.7';

    const apiParamter = {
      sCompanyCode: companyCode.trim(),
      sUserName: username.trim(),
      sPassWord: password,
      sSessionId: 'Mobile',
      sEmpCode: '',
      P_IP_ADDRESS: '',
      P_DEVICE_TYPE: Platform.OS,
      P_BROWSER_VERSION: Appversion,
      iCompId: '',
      P_BROWSER_TYPE:
        Platform?.constants?.Manufacturer +
        ' ' +
        Platform?.constants?.Model +
        ' ' +
        Platform?.constants?.Release,
      P_IMEI_NO: '',
      sDeviceName: 'Mobile',
    };

    try {
      const encrypted = await encryptUsingAES256(apiParamter).toString();

      const Config = {
        Url: 'User/GetLoginDeviceBinding',
        body: { sInput: encrypted },
        navigation,
      };

      const resp = await FetchHttpRequest(Config);

      if (resp?.statusCode === '400') {
        setLoading(false);
        Alert.alert('Login Error', resp?.statusMsg);
        return;
      }

      const decrypt = decryptUsingAES256(resp);
      const data = JSON.parse(decrypt);

      const compId = data?.data?.iCompId;
      setCompId(compId);
      setUserData(data);

      // fetch birthdays first, then navigate
      await fetchBirthdays(compId);

      // now ready to navigate
      setReadyToNavigate(true);
    } catch (error) {
      console.log('❌ Login error:', error);
      setLoading(false);
      Alert.alert('Login Failed', 'Something went wrong during login.');
    }
  };

  // 🔹 4️⃣ UI Section
  return (
    <KeyboardAvoidingView behavior="padding" style={{ flex: 1, backgroundColor: '#fff' }}>
      <ScrollView contentContainerStyle={{ flexGrow: 1, justifyContent: 'center', padding: 20 }}>
        <View style={{ alignItems: 'center' }}>
          <Image source={require('../assets/logo.png')} style={{ width: 120, height: 120 }} />
          <Text style={{ fontSize: 20, fontWeight: 'bold', marginVertical: 20 }}>Login</Text>

          {/* your input fields here */}
          {/* Example: */}
          {/* <TextInput value={username} onChangeText={setUsername} placeholder="Username" /> */}

          <Pressable
            onPress={doLogin}
            style={{
              marginTop: 20,
              backgroundColor: '#007bff',
              paddingVertical: 12,
              paddingHorizontal: 40,
              borderRadius: 8,
            }}>
            <Text style={{ color: '#fff', fontWeight: 'bold' }}>Login</Text>
          </Pressable>

          {loading && (
            <View style={{ marginTop: 20 }}>
              <ActivityIndicator size="large" color="#007bff" />
              <Text style={{ marginTop: 10, color: '#555' }}>Loading, please wait...</Text>
            </View>
          )}
        </View>
      </ScrollView>
    </KeyboardAvoidingView>
  );
};

export default LoginScreen;
