const LangSelectionPage = ({ navigation }) => {
  const { t, i18n } = useTranslation();
  const [currentLanguagecode, setLanguageCode] = useState('en');
  const [currentLanguage, setLanguage] = useState('');
  const [selectedLangIndex, setSelectedLangIndex] = useState(null);

  const LangData = [
    { title: 'English', val: 'en' },
    { title: 'العربية', val: 'ar' },
    { title: 'हिंदी', val: 'hn' },
    { title: 'ગુજરાતી', val: 'gu' },
    { title: 'ಕನ್ನಡ', val: 'kn' },
    { title: 'മലയാളം', val: 'mal' },
    { title: 'தமிழ்', val: 'tam' },
  ];

  useEffect(() => {
    const loadLanguage = async () => {
      const storedLang = await AsyncStorage.getItem("SelectedLang");
      if (storedLang) {
        setLanguageCode(storedLang);
        const index = LangData.findIndex(x => x.val === storedLang);
        setSelectedLangIndex(index);
        setLanguage(LangData[index]?.title);
      }
    };
    loadLanguage();
  }, []);

  const changeLanguage = async (value) => {
    try {
      await AsyncStorage.setItem("SelectedLang", value);
      await i18n.changeLanguage(value);
      setLanguageCode(value);
      navigation.navigate("MyWorkspace");
    } catch (err) {
      console.log("err:", err);
    }
  };
