const fillApplicationMenuList = async () => {
  try {
    const resp = await GetApplicationMenuList();
    const decrypted = decryptUsingAES256(resp);
    console.log('decrypted =======', decrypted);

    let parsed;

    // Parse the decrypted string only if it’s actually JSON text
    if (typeof decrypted === 'string') {
      try {
        parsed = JSON.parse(decrypted);
      } catch (err) {
        console.log('Error parsing decrypted JSON:', err);
        parsed = [];
      }
    } else {
      parsed = decrypted;
    }

    // Save parsed data directly to AsyncStorage
    await AsyncStorage.setItem('my_key', JSON.stringify(parsed));
    setLoading(false);
  } catch (error) {
    console.log('Error in fillApplicationMenuList:', error);
    setLoading(false);
  }
};
