  const GetTokenKey = async () => {
    if (username !== '' && password !== '') {
      setisBtnLoading(true);
      let Config = {
        Url: `MobLoginApi/GetTokenKey`,
      };
      const response = await TokenFetchHttpGet(Config);
      console.log('response GetTokenKey----> ', response);
      setisBtnLoading(false);
      if (response === 'Incorrect User Credentials') {
        setincorrectuser(response);
        setVisible(true);
        setIsValidLogin(true);
      } else if (response === 'Your Account has been locked') {
        setincorrectuser(response);
        setVisible(true);
      } else {
        if (response.KeyId !== null || response.Public_Key !== null || response.CaptchaImage !== null) {
          setKeyId(response.KeyId);
          await AsyncStorage.setItem('keyId', response.KeyId);
          setPublicKey(response.Public_Key);
          await AsyncStorage.setItem('publicKey', response.Public_Key);
          setCaptchaImage(response.CaptchaImage);
          await AsyncStorage.setItem('captchaImage', response.CaptchaImage);

          // console.log('response SessionId 222----> ', response.SessionId);
          // await AsyncStorage.setItem('LoginData', JSON.stringify(response));
          // await AsyncStorage.setItem(
          //   'SessionId',
          //   JSON.stringify(response.SessionId),
          // );
          encryptPassword(password, response.Public_Key);
        } else {
          ToastAndroid.show('Server Error!', ToastAndroid.SHORT);
        }
      }
    } else {
      setisBtnLoading(false);
      ToastAndroid.show(
        'Please enter username and password.',
        ToastAndroid.SHORT,
      );
    }
  };

  const encryptPassword = async (pass, pubKey) => {
    console.log("REACHING HEREEEEEEEEEEEEEE");
    if (pass !== '' && pubKey !== '') {
      console.log("SENDING THIS ++++", password + publicKey)
      setisBtnLoading(true);
      let Config = {
        Url: `MobLoginApi/EncryptData`,
        body: {
          Text: pass,
          PublicKey: pubKey
        }
      };
      const response = await TokenFetchHttpGet(Config);
      console.log('response encryptPassword----> ', response);
      setisBtnLoading(false);
      if (response === 'Incorrect User Credentials') {
        setincorrectuser(response);
        setVisible(true);
        setIsValidLogin(true);
      } else if (response === 'Your Account has been locked') {
        setincorrectuser(response);
        setVisible(true);
      } else {
        navigation.navigate('DrawerNavigation');

      }
    } else {
      setisBtnLoading(false);
      ToastAndroid.show(
        'Password and public key not received',
        ToastAndroid.SHORT,
      );
    }
  };
