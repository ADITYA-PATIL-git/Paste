const toggleDate = (item, newValue) => {
  setRevokeDate(prev => {
    const updated = prev.map(d =>
      d.LEAVE_DATE === item.LEAVE_DATE
        ? { ...d, checked: newValue ? 'Y' : 'N' }
        : d
    );

    const selectable = updated.filter(x =>
      !(x.SANDWICH == 1 || x.STATUS == '04' || x.STATUS == '05' || x.STATUS == '06')
    );

    setSelectAll(selectable.length > 0 && selectable.every(x => x.checked === 'Y'));

    const saveDates = selectable
      .filter(x => x.checked === 'Y')
      .map(x => ({ Dates: x.LEAVE_DATE }));

    setRevokeSaveDate(saveDates);
    setRevokeDataLength(saveDates.length);

    return updated;
  });
};
