const otpVerify = async (userId, sessionId, otp) => {
    console.log("REACHING HEREEEEEEEEEEEEEE");
    if (userId !== '' && sessionId !== '' && otp !== '') {
      setisBtnLoading(true);
      let Config = {
        Url: `MobLoginApi/ValidateLoginOTP`,
        method: 'POST',
        body: {
          UserID: userId,
          SessionId: sessionId,
          OTP: otp
        }
      };
      console.log('otpVerify Params----> ', Config.body);
      const response = await TokenFetchHttpGet(Config);
      console.log('response otpVerify----> ', response);
      setisBtnLoading(false);
      if (response === undefined) {
        ToastAndroid.show(
          response,
          ToastAndroid.SHORT,
        );
      } else {
        navigation.navigate('DrawerNavigation');
      }
    } else {
      setisBtnLoading(false);
      ToastAndroid.show(
        'Please Enter OTP',
        ToastAndroid.SHORT,
      );
    }
  };

export const TokenFetchHttpGet = async Config => {
  const UserId = await AsyncStorage.getItem('UserId')
  const SessionId = await AsyncStorage.getItem('SessionId')
  let Fetch = fetch(BaseURL + Config.Url, {
    method: Config.method ? Config.method : 'GET',
    body: Config.body ? JSON.stringify(Config.body) : null,    
    headers: {
      ClientID: 'qN5jT7+X5LI8S0aZT7Zvz50SvSEN9BoOtF3isKoryQtkM4HizLqrXw==',
      AuthID: UserId,
      AuthToken: SessionId,      
      'Content-Type': 'application/json',    
    },
  }).then(resp => resp.json()).catch(error => {
    console.log("error===FFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFF", error)
  });
  console.log('TokenFetchHttpGet RESPONSE--->', Fetch)
  return Fetch;
};
