import React, { useEffect, useState, useContext, useCallback } from 'react';
import { View, Text, Image, Dimensions, StyleSheet } from 'react-native';
import { useNavigation, getFocusedRouteNameFromRoute } from '@react-navigation/native';
import {
  createDrawerNavigator,
  DrawerContentScrollView,
  DrawerItem,
} from '@react-navigation/drawer';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import AsyncStorage from '@react-native-async-storage/async-storage';
import 'react-native-gesture-handler';

import { HeaderNotificationUser } from '../components/HeaderIcons';
import BottomTabNavigator from '../components/BottomTabNavigator';
import MyWorkSpace from './MyWorkSpace';
import DrawerButton from '../components/DrawerButton';
import DetailScreen from './DetailScreen';
import SearchScreen from './SearchScreen';
import { SearchContext } from '../context/SearchContext';
import { encryptUsingAES256 } from '../Encryption';
import { decryptUsingAES256 } from '../decryption';
import { FetchHttpRequest } from '../src/FetchHttpRequest';

const { width } = Dimensions.get('window');
const Drawer = createDrawerNavigator();
const Stack = createNativeStackNavigator();

/* ------------------- Utility Components ------------------- */
const SearchStack = () => (
  <Stack.Navigator screenOptions={{ headerShown: false }}>
    <Stack.Screen name="SearchScreen" component={SearchScreen} />
    <Stack.Screen name="DetailScreen" component={DetailScreen} />
  </Stack.Navigator>
);

const styles = StyleSheet.create({
  image1: {
    borderRadius: 100,
    height: width * 0.2,
    width: width * 0.2,
    marginLeft: width * 0.028,
  },
  drawerHeader: {
    fontSize: 16,
    fontWeight: 'bold',
    marginVertical: 10,
    marginLeft: 15,
  },
  noMenuText: {
    marginLeft: 20,
    color: 'gray',
  },
});

/* ------------------- Main Component ------------------- */
const RootScreen = ({ route }) => {
  const { userData, companyCode, bday, birthdayData, anniversaryData } = route.params;
  const navigation = useNavigation();
  const { setQuery } = useContext(SearchContext);

  const [loading, setLoading] = useState(false);
  const [profileData, setProfileData] = useState('');
  const [menuList, setMenuList] = useState([]);

  /* ------------------- Effects ------------------- */
  useEffect(() => {
    (async () => {
      await fetchMenuListFromStorage();
      await fillApplicationMenuList();
    })();
  }, []);

  useEffect(() => {
    if (userData?.data?.Original_Photo) {
      EmployeeProfilePhoto();
    }
  }, [userData]);

  /* ------------------- Data Fetchers ------------------- */
  const fetchMenuListFromStorage = useCallback(async () => {
    try {
      const jsonValue = await AsyncStorage.getItem('my_key');
      if (!jsonValue) return;

      let parsed = JSON.parse(jsonValue);
      // If double stringified
      if (typeof parsed === 'string') parsed = JSON.parse(parsed);

      const finalData = Array.isArray(parsed)
        ? parsed
        : parsed?.Table && Array.isArray(parsed.Table)
        ? parsed.Table
        : [];

      setMenuList(finalData);
      console.log('Menu list loaded:', finalData);
    } catch (error) {
      console.warn('Error loading menu list from AsyncStorage:', error);
      setMenuList([]);
    }
  }, []);

  const fillApplicationMenuList = useCallback(async () => {
    try {
      setLoading(true);
      const [CompanyName, UserIdData] = await Promise.all([
        AsyncStorage.getItem('COMPNAME'),
        AsyncStorage.getItem('UserId'),
      ]);

      if (!CompanyName || !UserIdData) {
        console.warn('Missing company or user data in storage.');
        setLoading(false);
        return;
      }

      const param = {
        sUserId: parseInt(UserIdData),
        sCompanyCode: CompanyName.trim(),
        P_ROLE_ID: 14697,
      };

      const encrypted = encryptUsingAES256(param).toString();
      const Config = {
        Url: 'Security/MenuMaster/FillMobileMenu',
        body: { sInput: encrypted },
      };

      const resp = await FetchHttpRequest(Config);
      const decrypted = decryptUsingAES256(resp);

      const data = typeof decrypted === 'string' ? JSON.parse(decrypted) : decrypted;
      await AsyncStorage.setItem('my_key', JSON.stringify(data));
      setMenuList(Array.isArray(data) ? data : data?.Table || []);
    } catch (error) {
      console.error('Error in fillApplicationMenuList:', error);
    } finally {
      setLoading(false);
    }
  }, []);

  const EmployeeProfilePhoto = useCallback(async () => {
    try {
      const param = {
        FileName: userData.data.sProfilePhoto,
        Identity: 'Profile',
        sCompanyCode: companyCode.trim(),
      };

      const encrypted = encryptUsingAES256(param).toString();
      const Config = { Url: `DownloadLogo`, body: { sInput: encrypted }, navigation };
      const resp = await FetchHttpRequest(Config);

      if (resp?.data) {
        setProfileData(resp.data);
        await AsyncStorage.setItem('base64', resp.data);
      } else {
        console.log('No profile image found.');
        setProfileData('');
      }
    } catch (error) {
      console.error('Error fetching profile photo:', error);
    }
  }, [userData, companyCode]);

  /* ------------------- Drawer Content ------------------- */
  const CustomDrawerContent = useCallback(
    props => (
      <DrawerContentScrollView {...props}>
        {profileData ? (
          <Image
            style={styles.image1}
            source={{ uri: `data:image/jpeg;base64,${profileData}` }}
          />
        ) : null}

        <Text style={styles.drawerHeader}>Menu</Text>

        {menuList.length > 0 ? (
          menuList.map((item, index) => (
            <DrawerItem
              key={index}
              label={item.MENU_NAME}
              onPress={() => props.navigation.navigate('HomeScreenTabs')}
            />
          ))
        ) : (
          <Text style={styles.noMenuText}>No menu items available</Text>
        )}

        <DrawerItem
          label="Home Screen"
          onPress={() => props.navigation.navigate('HomeScreenTabs')}
        />
        <DrawerItem
          label="Workspace"
          onPress={() => props.navigation.navigate('Workspace')}
        />
      </DrawerContentScrollView>
    ),
    [menuList, profileData],
  );

  /* ------------------- UI ------------------- */
  return (
    <Drawer.Navigator
      screenOptions={{
        drawerStatusBarAnimation: 'fade',
        drawerType: 'slide',
        overlayColor: 'transparent',
        sceneStyle: { backgroundColor: '#eef2fe' },
        headerTintColor: '#fff',
        headerShadowVisible: false,
        title: false,
        headerRight: () => <HeaderNotificationUser />,
        headerLeft: () => <DrawerButton />,
        freezeOnBlur: true,
      }}
      drawerContent={CustomDrawerContent}
    >
      <Drawer.Screen
        name="HomeScreenTabs"
        options={({ route }) => {
          const routeName = getFocusedRouteNameFromRoute(route) ?? 'Home';
          return {
            headerStyle: {
              backgroundColor: '#2860E7',
              borderBottomLeftRadius: routeName === 'Home' ? 0 : 14,
              borderBottomRightRadius: routeName === 'Home' ? 0 : 14,
            },
            headerTitle: routeName === 'Home' ? '' : routeName,
          };
        }}
      >
        {props => (
          <BottomTabNavigator
            {...props}
            userData={userData}
            profileData={profileData}
            bday={bday}
            birthdayData={birthdayData}
            anniversaryData={anniversaryData}
          />
        )}
      </Drawer.Screen>

      <Drawer.Screen
        name="Workspace"
        component={MyWorkSpace}
        options={{ headerStyle: { backgroundColor: '#2860E7' } }}
      />

      <Drawer.Screen
        name="SearchStack"
        component={SearchStack}
        options={{ headerStyle: { backgroundColor: '#2860E7' } }}
      />
    </Drawer.Navigator>
  );
};

export default RootScreen;
