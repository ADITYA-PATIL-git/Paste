async function downloadFile(pdfbase64, pdfname) {
  try {
    setIsLoading(true);

    // Request permission for Android below 10
    if (Platform.OS === 'android' && Platform.Version < 29) {
      const granted = await PermissionsAndroid.request(
        PermissionsAndroid.PERMISSIONS.WRITE_EXTERNAL_STORAGE,
      );

      console.log("Permission result:", granted);

      if (granted !== PermissionsAndroid.RESULTS.GRANTED) {
        Alert.alert('Permission denied', 'Cannot download the file');
        setIsLoading(false);
        return;
      }
    }

    const { dirs } = ReactNativeBlobUtil.fs;
    const path = `${dirs.DownloadDir}/${pdfname}.pdf`;

    ReactNativeBlobUtil.config({
      fileCache: true,
      addAndroidDownloads: {
        useDownloadManager: true,
        notification: true,
        mime: 'application/pdf',
        mediaScannable: true,
        title: `${pdfname}.pdf`,
        path: path,
        description: 'Downloading PDF...',
      },
    })
      .fetch('GET', `data:application/pdf;base64,${pdfbase64}`)
      .then(res => {
        console.log("File downloaded at:", res.path());
        FileViewer.open(res.path());
      })
      .catch(error => {
        console.log("Download error:", error);
        Alert.alert("Download Failed");
      })
      .finally(() => setIsLoading(false));
  } catch (error) {
    console.log("Error:", error);
    Alert.alert('Error', 'Failed to download.');
    setIsLoading(false);
  }
}
