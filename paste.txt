import React, { useEffect, useState } from "react";

import { Appbar } from 'react-native-paper';
import commonStyles from './CommanCSS';
import { useRoute } from '@react-navigation/native';
import Icon from 'react-native-vector-icons/MaterialIcons';
import Icon2 from 'react-native-vector-icons/MaterialCommunityIcons';
import { TextInput } from 'react-native-gesture-handler';
import ImagePopup from './BirthdayChatImagePopup';
import { Modal, View, Text, TouchableOpacity, SafeAreaView, FlatList, BackHandler, StyleSheet, Image, ScrollView } from "react-native";
import { FetchHttpRequest } from "./FetchHttp/FetchHttpRequest";
import { useIsFocused } from '@react-navigation/native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { decryptUsingAES256 } from './decryption';
import { encryptUsingAES256 } from './Encryptoion';
import { ActivityIndicator } from "react-native";
import { Alert } from "react-native";

const BirthdayAnniversaryChat = ({ navigation }) => {
    const route = useRoute();
    const { dataForChat, from } = route.params;
    const [popupVisible, setPopupVisible] = useState(false);
    const [selectedImage, setSelectedImage] = useState(null);
    const isFocused = useIsFocused();
    const [loading, setLoading] = useState(false);
    const [imageNamesList, setImageNamesList] = useState(null);
    const [base64List, setBase64List] = useState([]);
    const [birthdayText, setBirthdayText] = useState('');
    const [imageId, setImageId] = useState('');
    const RecipentUserId = dataForChat.USER_ID;



    const handleBackButton = () => {
        return true;
    };

    // const setImageData = async () => {
    //     const temp = [];
    //     for (const item of imageNamesList) {
    //         const data = await GetImageData(item.sIMMAGE_S3_BUCKET_FILE_NAME);
    //         const imageIdData = item.iIMMAGE_ID;
    //         if (data) temp.push({
    //             base64: data,
    //             imageId: imageIdData
    //         });
    //     }
    //     setBase64List(temp);
    // };
    const setImageData = async () => {
        if (!imageNamesList || imageNamesList.length === 0) return;

        const results = await Promise.all(
            imageNamesList.map(async item => {
                const data = await GetImageData(item.sIMMAGE_S3_BUCKET_FILE_NAME);
                return data ? { base64: data, imageId: item.iIMMAGE_ID } : null;
            })
        );

        setBase64List(results.filter(Boolean));
    };



    useEffect(() => {
        if (isFocused) {
            setLoading(true)
            getData().finally(() => { setLoading(false) });


            BackHandler.addEventListener('hardwareBackPress', handleBackButton);
            return () =>
                BackHandler.removeEventListener('hardwareBackPress', handleBackButton);

        }
    }, [isFocused]);


    const getData = async () => {
        setImageId('');
        setBase64List([]);
        setBirthdayText('');
        setSelectedImage('');

        const [
            CompanyID, UserName, EmployeeName,
            UserId, EmpId, SessionId, ConNo, CompanyName
        ] = await Promise.all([
            AsyncStorage.getItem('CompId'),
            AsyncStorage.getItem('UserName'),
            AsyncStorage.getItem('EmployeeName'),
            AsyncStorage.getItem('UserId'),
            AsyncStorage.getItem('EmployeeId'),
            AsyncStorage.getItem('SessionId'),
            AsyncStorage.getItem('ContactNO'),
            AsyncStorage.getItem('COMPNAME'),
        ]);



        await GetImageNames(CompanyName, UserId, UserName, SessionId);
    };

    async function GetImageNames() {

        var param = {
            sCompanyCode: CompanyName,
            iIMMAGE_ID: null,
            sIMMAGE_CATEGORY: from === "Birthday" ? "Birthday" : "Anniversary",
            sUserId: UserId,
            sUserName: UserName,
            sSessionId: SessionId,
            sModuleId: ""
            // sModuleId: 384
        };

        console.log("GetImages==param==", param)
        var encrypted = await encryptUsingAES256(param).toString();
        console.log("GetImages==encrypted==", encrypted)
        const Config = {
            Url: 'Dashboard/MailCelebration/GetAll',
            body: {
                sInput: encrypted,
            },
            navigation
        };
        try {
            FetchHttpRequest(Config).then(async resp => {
                // console.log("Get_employee_roster==",resp)
                var decrypt = decryptUsingAES256(resp);
                var d = JSON.parse(decrypt);
                // console.log("GetImages==res==", d)
                // console.log("CHECK THIS ++++++++ ", d[0].sIMMAGE_S3_BUCKET_FILE_NAME)
                setImageNamesList(d);

            });
        } catch (error) {

        } finally {
        }
    }

    async function GetImageData(sFileData) {

        var param = {
            FileName: sFileData,
            Identity: "Profile",
            sCompanyCode: CompanyName,
            sUserId: UserId,
            sUserName: UserName,
            sSessionId: SessionId,
            sModuleId: ""
        };

        var encrypted = await encryptUsingAES256(param).toString();
        const Config = {
            Url: 'DownloadLogo',
            body: { sInput: encrypted },
            navigation
        };

        try {
            const resp = await FetchHttpRequest(Config);
            return resp.data;
        } catch (error) {
            console.log("GetImageData error", error);
            return null;
        }
    }

    async function SendBirthdayWish() {
        let SMailValue = ''
        if (birthdayText && imageId) {
            SMailValue = `<HTML><BODY>${birthdayText}<P><BR></P></BODY></HTML>`
        }
        else if (birthdayText && imageId == '') {
            SMailValue = `<HTML><BODY>${birthdayText}</BODY></HTML>`
        }
        else if ((birthdayText == '' && imageId)) {
            SMailValue = `<HTML><BODY><P><BR></P></BODY></HTML>`
        }
        else {
            Alert.alert('Alert', 'Kindly Add Your Wish Or Add Image.')
            return
        }

        console.log('birthdayText -----', birthdayText)
        setLoading(true);
        var param = {
            sCompanyCode: CompanyName,
            iSENDER_USER_ID: UserId,
            iRECIPENT_USER_ID: RecipentUserId,
            iIMMAGE_ID: imageId,
            sCELEBRATION: from === "Birthday" ? "Birthday" : "Anniversary",
            sMAIL_MESSAGE: SMailValue,
            sUserId: UserId,
            sUserName: EmployeeName,
            sSessionId: SessionId,
            sModuleId: ""
            // sModuleId: 384
        };

        console.log("SendBirthdayWish==param==", param)

        var encrypted = await encryptUsingAES256(param).toString();
        console.log("SendBirthdayWish==encrypted==", encrypted)
        const Config = {
            Url: 'Dashboard/MailCelebrationInsert',
            body: {
                sInput: encrypted,
            },
            navigation
        };
        try {
            FetchHttpRequest(Config).then(async resp => {
                // console.log("Get_employee_roster==",resp)
                console.log("SendBirthdayWish==res==+++++++++++++++++", resp)
                if (resp?.statusCode) {
                    Alert.alert('Alert', resp?.statusMsg, [
                        {
                            text: 'OK',
                            onPress: () => {
                                navigation.navigate('Birthday');
                            },
                        },
                    ]);
                }
                else {
                    Alert.alert('Alert', 'Something Went Wrong...', [
                        {
                            text: 'OK',
                            onPress: () => {
                                navigation.navigate('Birthday');
                            },
                        },
                    ]);
                }

                setLoading(false);
            });
        } catch (error) {

        } finally {
        }
    }




    const handleImageSelect = (item) => {
        setSelectedImage(item.base64)
        setImageId(item.imageId)
        setPopupVisible(false);
    }

    return (
        <SafeAreaView style={{ flex: 1, }}>
            <Appbar.Header style={commonStyles.AppbarHeader}>
                <Appbar.BackAction
                    color="white"
                    onPress={() => {
                        navigation.navigate('Birthday');
                    }}
                />
                <Appbar.Content
                    title={
                        <Text style={commonStyles.AppbarTitle}>Birthday & Anniversary</Text>
                    }
                    fontWeight="bold"
                    color="white"

                />
            </Appbar.Header>

            {loading &&
                <Modal animationType="none" transparent={true}>
                    <View
                        style={{
                            flex: 1,
                            backgroundColor: 'rgba(0,0,0,0.6)',
                            justifyContent: 'center',
                            alignItems: 'center',
                        }}>
                        <ActivityIndicator size={'large'} color={'white'} />
                    </View>
                </Modal>
            }

            <Modal visible={popupVisible} transparent animationType="slide">
                <View style={{
                    flex: 1,
                    backgroundColor: "rgba(0,0,0,0.5)",
                    justifyContent: "center",
                    padding: 10,

                }}>
                    <View style={{ backgroundColor: "#fff", borderRadius: 10, padding: 20 }}>
                        <Text style={{ fontSize: 18, marginBottom: 10, color: '#000' }}>Select an Image</Text>

                        {base64List.length > 0 ? (
                            <FlatList
                                style={{ alignSelf: 'center' }}
                                data={base64List}
                                renderItem={({ item }) => {
                                    // console.log('Iten -------', item)
                                    return (
                                        <TouchableOpacity
                                            onPress={() => handleImageSelect(item)}
                                            style={{ padding: 10, }}
                                        >
                                            <Image
                                                source={{
                                                    uri: `data:image/jpeg;base64,${item.base64}`,
                                                }}
                                                // source={require('./images/dummy_profile_image.jpg')}
                                                style={{
                                                    marginHorizontal: 20,
                                                    width: 220,
                                                    height: 220,

                                                }}
                                            />
                                        </TouchableOpacity>
                                    )
                                }}
                            />) : (
                            <ActivityIndicator size={'large'} color={'green'} />
                        )}

                        <TouchableOpacity
                            onPress={() => setPopupVisible(false)}
                            style={{ marginTop: 15, padding: 10, backgroundColor: "#ddd" }}
                        >
                            <Text style={{ color: '#000', textAlign: "center" }}>Cancel</Text>
                        </TouchableOpacity>

                    </View>
                </View>
            </Modal>

            <ScrollView >
                <View style={styles.container2}>
                    <View style={{ flexDirection: 'row', alignItems: 'center' }}>
                        {dataForChat?.sThumbnailBase64 ==
                            'The specified key does not exist.' ||
                            dataForChat?.sThumbnailBase64 == '' ||
                            dataForChat?.sThumbnailBase64 == null ||
                            dataForChat?.sThumbnailBase64 == undefined ? (
                            <Image
                                // source={{ uri: `data:image/jpeg;base64,${item?.sThumbnailBase64}` }}
                                source={require('./images/dummy_profile_image.jpg')}
                                style={styles.image1}
                            />
                        ) : (
                            <Image
                                source={{
                                    uri: `data:image/jpeg;base64,${dataForChat?.sThumbnailBase64}`,
                                }}
                                // source={require('./images/dummy_profile_image.jpg')}
                                style={styles.image1}
                            />
                        )}

                        <View
                            style={{
                                flexDirection: 'column',
                                marginVertical: 15,
                                width: '60%',

                                marginRight: 2
                            }}>
                            <Text
                                numberOfLines={2}
                                style={[styles.EmpName, { color: '#fd8c8c' }]}>
                                {dataForChat?.EMP_NAME ? dataForChat?.EMP_NAME : '--'}

                            </Text>
                            <Text style={styles.EmpSubtext}>
                                {dataForChat?.DEPARTMENT_NAME ? dataForChat?.DEPARTMENT_NAME : '--'}
                            </Text>
                        </View>
                    </View>
                </View>
                <View style={{ marginHorizontal: 26, marginTop: 30, marginBottom: 20, flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center' }}>
                    <Text style={{ color: 'black', fontSize: 15 }}>
                        Your Wish
                    </Text>
                    <TouchableOpacity style={{ flexDirection: 'row', alignItems: 'center', }}
                        onPress={async () => {
                            setPopupVisible(true);

                            // If no image_names fetched yet → wait until GetImageNames finishes
                            if (!imageNamesList) return;

                            // If base64 list already fetched → don't re-fetch
                            if (base64List.length > 0) return;

                            setLoading(true);
                            await setImageData();
                            setLoading(false);
                        }}

                    >
                        <Icon2 name="plus" color="blue" size={24} />
                        <Text style={{ color: 'blue', fontSize: 15 }}>
                            Add Image
                        </Text>
                    </TouchableOpacity>

                </View>
                <View style={[styles.container2, { flex: 1, minHeight: 500 }]}>
                    <TextInput multiline style={{

                        flex: 1, padding: 10,
                        minHeight: 100,
                        textAlignVertical: 'top',
                        color: 'black'
                    }}
                        placeholderTextColor="grey"
                        placeholder="Write your wishes here"
                        value={birthdayText}
                        onChangeText={text => setBirthdayText(text)}
                    />
                    <View >
                        {/* <Text style={{ color: 'black' }}>
                            {selectedImage}
                        </Text> */}
                        <Image
                            source={{
                                uri: `data:image/jpeg;base64,${selectedImage}`,
                            }}
                            // source={require('./images/dummy_profile_image.jpg')}
                            style={{
                                height: 350,
                                width: 350
                            }}
                        />
                    </View>

                </View>
                <TouchableOpacity
                    style={{ alignSelf: 'center', backgroundColor: 'blue', paddingVertical: 10, marginTop: 10, paddingHorizontal: 20, borderRadius: 100 }}
                    onPress={() => SendBirthdayWish()}
                >
                    <Text style={{ fontSize: 16, color: '#fff' }}>
                        Send
                    </Text>
                </TouchableOpacity>

            </ScrollView>
        </SafeAreaView>
    )
}

const styles = StyleSheet.create({
    container2: {
        marginHorizontal: 20,
        marginTop: 10,
        width: '90%',
        backgroundColor: '#fff',
        borderWidth: 1,
        borderColor: '#9DB2BF',
        borderRadius: 8,
    },
    Todotext: {
        color: '#1a1a1a',
        fontSize: 16,
        fontFamily: 'Poppins SemiBold 600',
        paddingLeft: 20,
    },
    image1: {
        marginHorizontal: 20,
        width: 60,
        height: 60,
        borderRadius: 1000,
    },
    image: {
        marginHorizontal: 20,
        width: 60,
        height: 60,
        borderRadius: 1000,
    },
    iconstyle: {
        marginLeft: 4,
        marginVertical: 3,
    },
    iconBox: {
        width: '27%',
        backgroundColor: '#E55807',
        borderRadius: 5,
    },
    EmpName: {
        fontSize: 16,
        fontFamily: 'Poppins SemiBold 600',
        includeFontPadding: false,
    },
    EmpSubtext: {
        color: '#1a1a1a',
        fontSize: 12,
        fontFamily: 'Poppins SemiBold 600',
        includeFontPadding: false,
    },
});

export default BirthdayAnniversaryChat
