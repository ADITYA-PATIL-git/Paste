  static Future<Map<String, dynamic>> initPlatformState() async {
    var deviceData = <String, dynamic>{};

    try {
      if (Platform.isAndroid) {
        AndroidDeviceInfo androidInfo = await deviceInfoPlugin.androidInfo;
        deviceData = _readAndroidBuildData(androidInfo);
        final prefs = await SharedPreferences.getInstance();
        prefs.setString("deviceVersion", deviceData["version.release"]);
        prefs.setString("deviceName",
            deviceData["brand"] + deviceData["device"] + deviceData["model"]);
      } else {
        deviceData = {'Error': 'Unsupported platform'};
      }
    } catch (e) {
      deviceData = {'Error': 'Failed to get platform version: $e'};
    }

    return deviceData;
  }

  static Map<String, dynamic> _readAndroidBuildData(AndroidDeviceInfo build) {
    return <String, dynamic>{
      'version.release': build.version.release,
      'brand': build.brand,
      'device': build.device,
      'model': build.model,
    };
  }

  Future<void> _insertPageWise(uneqKey, pageID, pageName) async {
    final DateTime now = DateTime.now();
    final String formattedDate = DateFormat('yyyy-MM-dd HH:mm:ss').format(now);
    final prefs = await SharedPreferences.getInstance();
    var sid = prefs.getString('SID').toString();
    var deviceversionval = prefs.getString('deviceVersion').toString();
    var devicenameval = prefs.getString('deviceName').toString();
    const String url =
        "https://sit.osourceglobal.com/UAT_API_REWARD_LOYALTY/api/Login/InsertPageWiseLog";
    print("Response Data hr----------: $url");

    try {
      Map<String, String?> param = {
        "sOperationType": "PAGELOG",
        "sUserId": sid,
        "sBrowserType": "Mobile",
        "sActionId": "0",
        "sBrowserVersion": deviceversionval,
        "sSessionId": uneqKey,
        "sPageId": pageID,
        "sPageName": pageName,
        // "sPageId": sPageId,
        // "sPageName": sPageName,
        "sLoginTime": formattedDate,
        "sBrowserName": devicenameval
      };
      print("Response Data hr----------: $param");
      Map<String, String> headers = {"Content-type": "application/json"};
      var body = jsonEncode(param);
      var response = await http.post(
        Uri.parse(url),
        headers: headers,
        body: body,
      );
      var resp = convert.jsonDecode(response.body);
      print("Response Data----------: $resp");
      if (resp['statusCode'] == 200) {
        print("Response Data----------: $resp['statusMsg']");
      } else {
        print("Failed with status: ${resp['statusMsg']}");
      }
    } catch (e) {
      print("Error during GET request: $e");
    }

  Future<void> _insertPageWise(uneqKey, pageID, pageName) async {
    final DateTime now = DateTime.now();
    final String formattedDate = DateFormat('yyyy-MM-dd HH:mm:ss').format(now);
    final prefs = await SharedPreferences.getInstance();
    var sid = prefs.getString('SID').toString();
    var deviceversionval = prefs.getString('deviceVersion').toString();
    var devicenameval = prefs.getString('deviceName').toString();
    const String url =
        "https://sit.osourceglobal.com/UAT_API_REWARD_LOYALTY/api/Login/InsertPageWiseLog";
    print("Response Data hr----------: $url");

    try {
      Map<String, String?> param = {
        "sOperationType": "PAGELOG",
        "sUserId": sid,
        "sBrowserType": "Mobile",
        "sActionId": "0",
        "sBrowserVersion": deviceversionval,
        "sSessionId": uneqKey,
        "sPageId": pageID,
        "sPageName": pageName,
        // "sPageId": sPageId,
        // "sPageName": sPageName,
        "sLoginTime": formattedDate,
        "sBrowserName": devicenameval
      };
      print("Response Data hr----------: $param");
      Map<String, String> headers = {"Content-type": "application/json"};
      var body = jsonEncode(param);
      var response = await http.post(
        Uri.parse(url),
        headers: headers,
        body: body,
      );
      var resp = convert.jsonDecode(response.body);
      print("Response Data----------: $resp");
      if (resp['statusCode'] == 200) {
        print("Response Data----------: $resp['statusMsg']");
      } else {
        print("Failed with status: ${resp['statusMsg']}");
      }
    } catch (e) {
      print("Error during GET request: $e");
    }
  }

class HexColor extends Color {
  static int _getColorFromHex(String hexColor) {
    hexColor = hexColor.toUpperCase().replaceAll("#", "");
    if (hexColor.length == 6) {
      hexColor = "FF$hexColor";
    }
    return int.parse(hexColor, radix: 16);
  }

  HexColor(final String hexColor) : super(_getColorFromHex(hexColor));
}
