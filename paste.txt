import React, { useState } from 'react';
import { View, Text, Modal, Pressable } from 'react-native';
import { Calendar } from 'react-native-calendars';

export default function InstantCalendarPicker() {
  const [selected, setSelected] = useState('');
  const [open, setOpen] = useState(false);

  return (
    <View style={{ padding: 16 }}>
      <Pressable
        onPress={() => setOpen(true)}
        style={{ padding: 12, backgroundColor: '#eee', borderRadius: 8 }}
      >
        <Text>{selected ? `Selected: ${selected}` : 'Pick a date'}</Text>
      </Pressable>

      <Modal visible={open} animationType="slide" transparent={true}>
        <View style={{
          flex: 1, justifyContent: 'center', backgroundColor: 'rgba(0,0,0,0.4)',
          padding: 16
        }}>
          <View style={{ backgroundColor: 'white', borderRadius: 12, padding: 12 }}>
            <Calendar
              onDayPress={(day) => {
                setSelected(day.dateString); // instant selection
                setOpen(false);
              }}
              markedDates={{
                [selected]: { selected: true, disableTouchEvent: true }
              }}
            />

            <Pressable onPress={() => setOpen(false)} style={{ marginTop: 8, alignSelf: 'center' }}>
              <Text style={{ color: '#007aff' }}>Cancel</Text>
            </Pressable>
          </View>
        </View>
      </Modal>
    </View>
  );
}
