useEffect(() => {
  const translateAnnouncements = async () => {
    if (!AnnouncementData || AnnouncementData.length === 0) return;

    try {
      const translated = await Promise.all(
        AnnouncementData.map(async (item) => {
          const translatedTitle = await translate(item.title);
          const translatedDesc = await translate(item.description);

          return {
            ...item,
            title: translatedTitle,
            description: translatedDesc,
          };
        })
      );

      setTranslatedAnnouncements(translated);
    } catch (error) {
      console.log("Announcement translation error", error);
    }
  };

  translateAnnouncements();
}, [AnnouncementData]);
