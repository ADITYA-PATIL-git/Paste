export const TokenFetchHttpGet = async Config => {
  try {
    const Fecth = await fetch(
      `https://sit.osourceglobal.com/INT_UAT_BAJAJFINSERV_AMC_API/api/${Config.Url}`,
      {
        method: Config.method ? Config.method : 'GET',
        body: Config.body ? JSON.stringify(Config.body) : null,
        headers: {
          'Content-Type': 'application/json',
          ClientID: 'qN5jT7+X5LI8S0aZT7Zvz50SvSEN9BoOtF3isKoryQtkM4HizLqrXw==',
        },
      }
    );

    const text = await Fecth.text();
    console.log('Response (text) -> ', text);
    return text; // <-- IMPORTANT

  } catch (error) {
    console.log('Fetch error -> ', error);
    throw error;
  }
};
