import 'dart:convert';

factory Album.fromJson(Map<String, dynamic> json) {
  dynamic data;

  // Decode only if it's a valid non-empty string
  if (json['data'] is String && (json['data'] as String).isNotEmpty) {
    data = jsonDecode(json['data']);
  } else if (json['data'] is Map<String, dynamic>) {
    data = json['data'];
  } else {
    data = {}; // fallback empty map
  }

  final img = data['sBase64'] ?? '';

  return Album(
    Name: data['sName'] ?? '',
    EmailId: data['sEmail'] ?? '',
    Mobile: data['sMobile'] ?? '',
    DOB: data['sDob'] ?? '',
    img64: img,
    Address: data['sAddress'] ?? '',
    cardValidity: data['sValidityCard'] ?? '',
    CardNo: data['sCardNumber'] ?? '',
    CardType: data['sMemberShipName'] ?? '',
    MembershipName: data['sMemberShipName'] ?? '',
    sMembershipType: data['sMembershipType'] ?? '',
  );
}
