      iSubMenuId: 204,
      sCompanyCode: CompanyName,
      sUserId: UserId,
      sUserName: UserName,
      sSessionId: SessionId,

      Url: 'WorkflowSetup/TransactionStatus/GetStatus',
      body: {
        sInput: encrypted,
      },
      navigation

        sInput: encrypted,

      Url: 'LNASetup/Leave/LeaveSubmit',
