              color="#1a1a1a"
              size={20}
            />
            <View
              style={{
                height: '50%',
                width: 1,
                backgroundColor: '#bbbbbb',
              }}></View>
            <TextInput
              style={[commonStyles.TextInput, {fontSize: 16, flex: 1}]}
              placeholder={mobileLogin ? 'Enter mobile no.' : 'Enter Username'}
              placeholderTextColor="#818181ff"
              onChangeText={handleUsernameChange}
              value={username}
              maxLength={50}
            />
          </View>
        </View>
        {!mobileLogin && (
          <View style={[commonStyles.formtextiputwidth, {marginBottom: 10}]}>
            <View
              style={[commonStyles.labelContainer, {backgroundColor: '#fff'}]}>
              <Text style={commonStyles.inputlabelstyle}>Password</Text>
            </View>

            <View
              style={[
                commonStyles.inputContainer,
                {height: 50, flexDirection: 'row', alignItems: 'center'},
              ]}>
              <Icon1
                style={{paddingHorizontal: 15}}
                name="lock"
                color="#1a1a1a"
                size={20}
              />
              <View
                style={{
                  height: '50%',
                  width: 1,
                  backgroundColor: '#bbbbbb',
                }}></View>
              <TextInput
                style={[commonStyles.TextInput, {fontSize: 16, flex: 1}]}
                placeholder="Enter Password"
                placeholderTextColor="#818181ff"
                onChangeText={handlePasswordChange}
                secureTextEntry={hidePass ? true : false}
                value={password}
                maxLength={50}
              />
              <TouchableOpacity
                onPress={() => {
                  setHidePass(!hidePass);
                }}>
                <Icon
                  style={{paddingRight: 15}}
                  name={hidePass ? 'eye-off-outline' : 'eye-outline'}
                  color="#1a1a1a"
                  size={25}
                />
              </TouchableOpacity>
            </View>
          </View>
        )}
        {isValidLogin && (
          <View>
            <Text style={{color: 'red'}}>Invalid Username and password</Text>
          </View>
        )}
        <View
          style={{
            flexDirection: 'row',
            alignItems: 'center',
            width: '100%',
            justifyContent: 'space-between',
          }}>
          <View style={commonStyles.checkboxrow}>
            <CheckBox
              tintColors={{
                false: 'black',
                true: '#2C61F3',
              }}
              value={ssoCheckbox}
              onValueChange={value => checkboxHandler(value)}
              disabled={isBtnLoading || biometricLoading}
            />
            <Text
              style={{
                fontSize: 13,
                color: '#000',
                includeFontPadding: false,
                fontFamily: 'Poppins Regular 400',
                paddingLeft: 4,
              }}>
              Stay logged in?
            </Text>
          </View>
          <TouchableOpacity
            onPress={async () => {
              try {
                const newValue = !mobileLogin;
                setMobileLogin(newValue);
                setUsername('');
                setPassword('');
                await AsyncStorage.setItem(
                  'mobileLogin',
                  JSON.stringify(newValue),
                );
                await AsyncStorage.removeItem('AsyncUserName');
                await AsyncStorage.removeItem('UserId');
                await AsyncStorage.removeItem('SessionId');
                await AsyncStorage.removeItem('biometricEnabled');
                await AsyncStorage.removeItem('password');
                biometricLockRef.current = true;
              } catch (e) {
                console.log('Failed to save login mode', e);
              }
            }}>
            <Text style={[commonStyles.welcomebluetext, {marginRight: 10}]}>
              {mobileLogin ? 'login with username' : 'login with mobile no.'}
            </Text>
          </TouchableOpacity>
        </View>
        <TouchableOpacity
          style={[commonStyles.SaveButton, {marginTop: 5}]}
          disabled={isBtnLoading || biometricLoading}
          onPress={handleLogin}>
          {isBtnLoading || biometricLoading ? (
            <View style={commonStyles.loadingstyle}>
              <ActivityIndicator size={20} color="white" />
            </View>
          ) : (
            <Text style={commonStyles.SaveTextstyle}>
              {/* {captchaImage ? 'Send OTP' : 'Log In'}
               */}
              Login
            </Text>
          )}
        </TouchableOpacity>
      </View>
      <View
        style={{
          flex: 1,
          flexDirection: 'row',
          justifyContent: 'space-evenly',
          marginTop: 20,
        }}>
        <TouchableOpacity
          onPress={CheckCanDoForgetPassword}
          disabled={isBtnLoading || biometricLoading}>
          <Text style={[commonStyles.welcomebluetext]}>Forgot Password?</Text>
        </TouchableOpacity>
        <TouchableOpacity
          onPress={CheckCanDoResetPassword}
          disabled={isBtnLoading || biometricLoading}>
          <Text style={[commonStyles.welcomebluetext]}>Reset Password?</Text>
        </TouchableOpacity>
      </View>

      {/* <View
        style={{
          flexDirection: 'row',
          alignItems: 'center',
          justifyContent: 'space-evenly',
        }}>
        <TouchableOpacity
          onPress={CheckCanDoResetPassword}
          disabled={isBtnLoading || biometricLoading}>
          <Text style={[commonStyles.welcomebluetext]}>Reset Password?</Text>
        </TouchableOpacity>
      </View> */}
    </ScrollView>
  );
}
