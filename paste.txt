const { width } = Dimensions.get('window');

function Documents({ navigation }) {
    const [isLoading, setIsLoading] = useState(true);
    const [Birthdata, setBirth] = useState([]);
    const [Annidata, setAnni] = useState([]);
    const [documentData, setDocumentData] = useState([]);
    const isFocused = useIsFocused();
    const Tab = createMaterialTopTabNavigator();
    const { t } = useTranslation('LeaveTranslation')
    const Loader = useRef('')

    useEffect(() => {
        if (isFocused) {
            getData();
            console.log('document data-----------', documentData)
        }
    }, [isFocused]);


    const getData = async () => {
        CompanyID = await AsyncStorage.getItem('CompId');
        UserName = await AsyncStorage.getItem('UserName');
        EmpName = await AsyncStorage.getItem('EmployeeName');
        UserId = await AsyncStorage.getItem('UserId');
        EmpId = await AsyncStorage.getItem('EmployeeId');
        SessionId = await AsyncStorage.getItem('SessionId');
        CompanyName = await AsyncStorage.getItem('COMPNAME');

        getDocuments();
    };

    async function downloadFile(pdfbase64, pdfname) {
        try {


            if (Platform.OS === 'android' && Platform.Version < 29) {
                const granted = await PermissionsAndroid.request(
                    PermissionsAndroid.PERMISSIONS.WRITE_EXTERNAL_STORAGE,
                );
                if (granted !== PermissionsAndroid.RESULTS.GRANTED) {
                    Alert.alert('Permission required');
                    setIsLoading(false);
                    return;
                }
            }

            const { fs } = ReactNativeBlobUtil;
            const { LegacyDownloadDir } = fs.dirs;
            const filePath = `${LegacyDownloadDir}/${pdfname}`;

            await fs.writeFile(filePath, pdfbase64, 'base64');

            PushNotification.localNotification({
                title: 'Download Complete',
                message: `${pdfname} has been saved in downloads folder`,
                channelId: '1',
            });
            console.log("Saved file to:", filePath);

            FileViewer.open(filePath)
                .catch(() => Alert.alert('Error', 'Unable to open file'))
                .finally(() => setIsLoading(false));

        } catch (e) {
            console.log("Error:", e);
            Alert.alert("Download failed");
            setIsLoading(false);
        }
    }


    let getDocuments = async function () {
        console.log("getDocuments-------")
        setIsLoading(true);
        param = {
            sCompanyCode: CompanyName,
            sDateFormat: 'DD-MM-YYYY',
            iEmpId: EmpId,
            sUserId: UserId,
            sUserName: UserName,
            sSessionId: SessionId,
            sModuleId: 47,
        };
        var encrypted = encryptUsingAES256(param).toString();
        let Config = {
            Url: 'Document/GetDocumentTab',
            body: {
                sInput: encrypted,
            },
            navigation
        };
        FetchHttpRequest(Config).then(async resp => {
            setIsLoading(false);

            var decrypt = decryptUsingAES256(resp);
            // console.log('its here', decrypt)
            var d = JSON.parse(decrypt);
            // console.log('JJJJJJJJJJJJ-------------------', d)
            setDocumentData(d)




            // if (d != '') {
            //     setLHistory(d?.data);
            // } else {
            //     Toast.show('Data Not Found');
            // }
            // setLHistory(d?.data);
        }).catch(error => {
            setIsLoading(false);
        })
    };

    return (
        <SafeAreaView style={{ flex: 1 }}>
            {isLoading ? (
                <ActivityIndicator
                    size="large"
                    color={'#2f66ea'}
                    style={{ height: 200 }}
                />

            ) : (
                Array.isArray(documentData) && documentData.length > 0 ? (
                    <View style={{ padding: 10, flex: 1 }}>
                        {documentData.map((item, index) => (
                            <View
                                style={{
                                    flexDirection: 'row',
                                    alignItems: 'center',
                                    justifyContent: 'space-between',
                                    padding: 10,
                                    marginBottom: 10,
                                    borderRadius: 10,
                                    borderWidth: 1.5,
                                    borderColor: '#bebebeff'
                                }}
                                key={index}
                            >
                                <View style={{ width: '90%' }}>
                                    <Text style={{ color: 'black', fontSize: 16, fontWeight: 'bold', marginBottom: 2 }}>
                                        {item?.sDocName ?? 'Unnamed Document'}
                                    </Text>
                                    <Text style={{ color: '#414141ff', fontSize: 14, marginBottom: 2 }}>
                                        {item?.sDocumentType ?? 'Unknown Type'}
                                    </Text>
                                    <Text style={{ color: '#727272ff', fontSize: 14 }}>
                                        {item?.p_S3_BUCKET_FILE_NAME ?? 'No filename'}
                                    </Text>
                                </View>

                                <TouchableOpacity
                                    disabled={!item?.sThumbnailBase64 || !item?.p_S3_BUCKET_FILE_NAME}
                                    onPress={() => {
                                        console.log('Downloading:', item?.p_S3_BUCKET_FILE_NAME)
                                        downloadFile(
                                            item?.sThumbnailBase64 ?? '',
                                            item?.p_S3_BUCKET_FILE_NAME ?? 'Unknown.pdf'
                                        )
                                    }}
                                >
                                    <Icon name="download" color="#000" size={30} />
                                </TouchableOpacity>
                            </View>
                        ))}
                    </View>
                ) : (
                    !isLoading && (
                        <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center' }}>
                            <Text style={{ color: '#444' }}>No documents found</Text>
                        </View>
                    )
                )

            )
            }
        </SafeAreaView >
    );
}
