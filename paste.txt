factory Album.fromJson(Map<String, dynamic> json) {
  // Decode nested string if needed
  final data = json['data'] is String ? jsonDecode(json['data']) : json['data'];

  // Use default image if null
  final img64 = data['sBase64'] ?? sBase64;

  return Album(
    Name: data['sName'] ?? '',
    EmailId: data['sEmail'] ?? '',
    Mobile: data['sMobile'] ?? '',
    DOB: data['sDob'] ?? '',
    img64: img64,
    Address: data['sAddress'] ?? '',
    cardValidity: data['sValidityCard'] ?? '',
    CardNo: data['sCardNumber'] ?? '',
    CardType: data['sMemberShipName'] ?? '',
    MembershipName: data['sMemberShipName'] ?? '',
    sMembershipType: data['sMembershipType'] ?? '',
  );
}
