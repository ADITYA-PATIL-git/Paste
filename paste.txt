const BirthdayChat = ({ navigation }) => {
    const route = useRoute();
    const { dataForChat } = route.params;
    const [popupVisible, setPopupVisible] = useState(false);
    const [selectedImage, setSelectedImage] = useState('');
    const isFocused = useIsFocused();
    const [loading, setLoading] = useState(false);
    const [imageNamesList, setImageNamesList] = useState(null);
    const [base64List, setBase64List] = useState([]);



    const handleBackButton = () => {
        return true;
    };

    const setImageData = async () => {
        const temp = [];
        for (const item of imageNamesList) {

            const data = await GetImageData(item.sIMMAGE_S3_BUCKET_FILE_NAME);
            if (data) temp.push(data);

        }
        setBase64List(temp);
    };

    useEffect(() => {
        if (imageNamesList) {
            setImageData();
        }
    }, [imageNamesList]);

    useEffect(() => {
        if (isFocused) {
            //   setAttachmentFag('')          
            setLoading(true);
            getData();
            setLoading(false);
            BackHandler.addEventListener('hardwareBackPress', handleBackButton);
            return () => {
                BackHandler.removeEventListener('hardwareBackPress', handleBackButton);
            };
        }
    }, [isFocused]);


    const getData = async () => {
        CompanyID = await AsyncStorage.getItem('CompId');
        UserName = await AsyncStorage.getItem('UserName');
        UserId = await AsyncStorage.getItem('UserId');
        EmpId = await AsyncStorage.getItem('EmployeeId');
        SessionId = await AsyncStorage.getItem('SessionId');
        ConNo = await AsyncStorage.getItem('ContactNO');
        CompanyName = await AsyncStorage.getItem('COMPNAME');
        var SecondLeaveID = 'S';
        await AsyncStorage.setItem('SecondLeaveID', SecondLeaveID);
        var LeaveDayId = 'F';
        await AsyncStorage.setItem('LeaveDayId', LeaveDayId);
        // LeaveType();
        GetImageNames();
    };

    async function GetImageNames() {
        setLoading(true);
        var param = {
            sCompanyCode: CompanyName,
            iIMMAGE_ID: null,
            sIMMAGE_CATEGORY: "Birthday",
            sUserId: UserId,
            sUserName: UserName,
            sSessionId: SessionId,
            sModuleId: ""
            // sModuleId: 384
        };

        console.log("GetImages==param==", param)
        var encrypted = await encryptUsingAES256(param).toString();
        console.log("GetImages==encrypted==", encrypted)
        const Config = {
            Url: 'Dashboard/MailCelebration/GetAll',
            body: {
                sInput: encrypted,
            },
            navigation
        };
        try {
            FetchHttpRequest(Config).then(async resp => {
                // console.log("Get_employee_roster==",resp)
                var decrypt = decryptUsingAES256(resp);
                var d = JSON.parse(decrypt);
                console.log("GetImages==res==", d)
                console.log("CHECK THIS ++++++++ ", d[0].sIMMAGE_S3_BUCKET_FILE_NAME)
                setImageNamesList(d);
                setLoading(false);
            });
        } catch (error) {

        } finally {
        }
    }

    async function GetImageData(sFileData) {
        setLoading(true);
        var param = {
            FileName: sFileData,
            Identity: "Profile",
            sCompanyCode: CompanyName,
            sUserId: UserId,
            sUserName: UserName,
            sSessionId: SessionId,
            sModuleId: ""
        };

        var encrypted = await encryptUsingAES256(param).toString();
        const Config = {
            Url: 'DownloadLogo',
            body: { sInput: encrypted },
            navigation
        };

        try {
            const resp = await FetchHttpRequest(Config);
            setLoading(false);
            return resp.data;
        } catch (error) {
            console.log("GetImageData error", error);
            return null;
        }
    }


    const handleImageSelect = (img) => {
        setSelectedImage(img)
        setPopupVisible(false);
    }

    return (
        <SafeAreaView style={{ flex: 1, }}>
            <Appbar.Header style={commonStyles.AppbarHeader}>
                <Appbar.BackAction
                    color="white"
                    onPress={() => {
                        navigation.navigate('Birthday');
                    }}
                />
                <Appbar.Content
                    title={
                        <Text style={commonStyles.AppbarTitle}>Birthday & Anniversary</Text>
                    }
                    fontWeight="bold"
                    color="white"

                />
            </Appbar.Header>
            <View style={styles.container2}>
                <View style={{ flexDirection: 'row', alignItems: 'center' }}>
                    {dataForChat?.sThumbnailBase64 ==
                        'The specified key does not exist.' ||
                        dataForChat?.sThumbnailBase64 == '' ||
                        dataForChat?.sThumbnailBase64 == null ||
                        dataForChat?.sThumbnailBase64 == undefined ? (
                        <Image
                            // source={{ uri: `data:image/jpeg;base64,${item?.sThumbnailBase64}` }}
                            source={require('./images/dummy_profile_image.jpg')}
                            style={styles.image1}
                        />
                    ) : (
                        <Image
                            source={{
                                uri: `data:image/jpeg;base64,${dataForChat?.sThumbnailBase64}`,
                            }}
                            // source={require('./images/dummy_profile_image.jpg')}
                            style={styles.image1}
                        />
                    )}

                    <View
                        style={{
                            flexDirection: 'column',
                            marginVertical: 15,
                            width: '60%',
                            backgroundColor: 'pink',
                            marginRight: 2
                        }}>
                        <Text
                            numberOfLines={2}
                            style={[styles.EmpName, { color: '#fd8c8c' }]}>
                            {dataForChat?.EMP_NAME ? dataForChat?.EMP_NAME : '--'}

                        </Text>
                        <Text style={styles.EmpSubtext}>
                            {dataForChat?.DEPARTMENT_NAME ? dataForChat?.DEPARTMENT_NAME : '--'}
                        </Text>
                    </View>
                </View>
            </View>
            <View style={{ marginHorizontal: 24, marginTop: 20, marginBottom: 20, flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center' }}>
                <Text style={{ color: 'black', fontSize: 15 }}>
                    Your Wish
                </Text>
                <TouchableOpacity style={{ flexDirection: 'row', alignItems: 'center', }}
                    onPress={() => setPopupVisible(true)}
                >
                    <Icon2 name="plus" color="blue" size={24} />
                    <Text style={{ color: 'blue', fontSize: 15 }}>
                        Add Image
                    </Text>
                </TouchableOpacity>

            </View>
            {loading ? (
                <ActivityIndicator size="large" color="#1a55e3" />
            ) : (
                // <ImagePopup
                //     visible={popupVisible}
                //     onClose={() => {
                //         setPopupVisible(false);
                //     }}
                //     onSelect={handleImageSelect}
                // />
                <Modal visible={popupVisible} transparent animationType="slide">
                    <View style={{
                        flex: 1,
                        backgroundColor: "rgba(0,0,0,0.5)",
                        justifyContent: "center",
                        padding: 20
                    }}>
                        <View style={{ backgroundColor: "#fff", borderRadius: 10, padding: 20 }}>
                            <Text style={{ fontSize: 18, marginBottom: 10 }}>Select an Image</Text>

                            <FlatList
                                data={base64List}
                                renderItem={({ item }) => (
                                    <TouchableOpacity
                                        onPress={() => handleImageSelect(item)}
                                        style={{ padding: 10, borderBottomWidth: 1 }}
                                    >
                                        <Image
                                            source={{
                                                uri: `data:image/jpeg;base64,${item}`,
                                            }}
                                            // source={require('./images/dummy_profile_image.jpg')}
                                            style={styles.image1}
                                        />
                                    </TouchableOpacity>
                                )}
                            />

                            <TouchableOpacity
                                onPress={() => setPopupVisible(false)}
                                style={{ marginTop: 15, padding: 10, backgroundColor: "#ddd" }}
                            >
                                <Text style={{ textAlign: "center" }}>Cancel</Text>
                            </TouchableOpacity>

                        </View>
                    </View>
                </Modal>
            )

            }
            <ScrollView style={{ flex: 1, }}>
                <View style={[styles.container2, { minHeight: 500 }]}>
                    <TextInput multiline style={{ backgroundColor: 'pink', flex: 1, padding: 10, textAlignVertical: 'top', color: 'black' }} />
                    <View style={{ height: 200 }}>
                        <Text style={{ color: 'black' }}>
                            {selectedImage}
                        </Text>
                    </View>

                </View>

            </ScrollView>





        </SafeAreaView>
    )
}
