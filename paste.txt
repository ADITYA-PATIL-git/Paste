export const TokenFetchHttpGet = async (Config) => {
  const UserId = await AsyncStorage.getItem('UserId');
  const SessionId = await AsyncStorage.getItem('SessionId');

  try {
    const response = await fetch(BaseURL + Config.Url, {
      method: Config.method || 'GET',
      headers: {
        ClientID: 'qN5jT7+X5LI8S0aZT7Zvz50SvSEN9BoOtF3isKoryQtkM4HizLqrXw==',
        AuthID: UserId,
        AuthToken: SessionId,
        'Content-Type': 'application/json',
      },
      body:
        Config.method === 'POST' && Config.body
          ? JSON.stringify(Config.body)
          : undefined,
    });

    const text = await response.text();

    // Safely parse JSON
    let parsed;
    try {
      parsed = text ? JSON.parse(text) : {};
    } catch {
      parsed = { OutcomeDetail: text };
    }

    // Attach HTTP info but DO NOT nest data
    if (!response.ok) {
      parsed._httpError = true;
      parsed._httpStatus = response.status;
      parsed._httpMessage = `HTTP_${response.status}`;
    }

    return parsed;

  } catch (error) {
    console.log('TokenFetchHttpGet error ===', error);
    return {
      OutcomeDetail: 'Network error. Please try again.',
      _networkError: true,
    };
  }
};
