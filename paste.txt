import React from "react";
import { Modal, View, Text, TouchableOpacity, FlatList } from "react-native";

const images = ["cat.png", "dog.png", "lion.png"]; // sample list

export default function ImagePopup({ visible, onClose, onSelect }) {
  return (
    <Modal visible={visible} transparent animationType="slide">
      <View style={{
        flex: 1,
        backgroundColor: "rgba(0,0,0,0.5)",
        justifyContent: "center",
        padding: 20
      }}>
        <View style={{ backgroundColor: "#fff", borderRadius: 10, padding: 20 }}>
          <Text style={{ fontSize: 18, marginBottom: 10 }}>Select an Image</Text>

          <FlatList
            data={images}
            keyExtractor={(item) => item}
            renderItem={({ item }) => (
              <TouchableOpacity
                onPress={() => onSelect(item)}
                style={{ padding: 10, borderBottomWidth: 1 }}
              >
                <Text>{item}</Text>
              </TouchableOpacity>
            )}
          />

          <TouchableOpacity
            onPress={onClose}
            style={{ marginTop: 15, padding: 10, backgroundColor: "#ddd" }}
          >
            <Text style={{ textAlign: "center" }}>Cancel</Text>
          </TouchableOpacity>

        </View>
      </View>
    </Modal>
  );
}
