useEffect(() => {
  const fetchMenuList = async () => {
    try {
      const jsonValue = await AsyncStorage.getItem('my_key');
      if (jsonValue) {
        const parsed = JSON.parse(jsonValue);
        console.log('Loaded menu list:', parsed);

        if (Array.isArray(parsed)) {
          setMenuList(parsed);
        } else {
          console.log('Unexpected menu data format');
          setMenuList([]);
        }
      }
    } catch (error) {
      console.log('Error reading menu list:', error);
    }
  };

  fetchMenuList();
}, []);
