<CheckBox
  tintColors={{ true: '#1a55e3' }}
  value={selectAll}
  disabled={allDisabled}
  onValueChange={(newValue) => {

    setSelectAll(newValue);

    // Update only selectable items
    setRevokeDate(prev => 
      prev.map(item => {
        const isDisabled =
          item?.SANDWICH == 1 ||
          item?.STATUS == '06' ||
          item?.STATUS == '05' ||
          item?.STATUS == '04';

        return {
          ...item,
          checked: isDisabled ? item.checked : (newValue ? 'Y' : 'N')
        };
      })
    );

    if (newValue) {
      // Save only selectable ones
      const updated = RevokeDate
        .filter(item =>
          !(item?.SANDWICH == 1 ||
            item?.STATUS == '06' ||
            item?.STATUS == '05' ||
            item?.STATUS == '04')
        )
        .map(item => ({ Dates: item?.LEAVE_DATE }));

      setRevokeSaveDate(updated);
      setRevokeDataLength(updated.length);
    } else {
      setRevokeSaveDate([]);
      setRevokeDataLength(0);
    }
  }}
/>
