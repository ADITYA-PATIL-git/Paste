import React, { useState } from 'react';
import {
  View,
  Text,
  TouchableOpacity,
  StyleSheet,
  ScrollView,
} from 'react-native';

const RadioButtonTree = ({
  title = 'Select Subject',
  data = [],
  defaultExpanded = '',
  defaultSelected = '',
  onSelect,
}) => {
  const [expanded, setExpanded] = useState(defaultExpanded);
  const [selected, setSelected] = useState(defaultSelected);

  const toggleExpand = itemTitle => {
    setExpanded(prev => (prev === itemTitle ? '' : itemTitle));
  };

  const handleSelect = child => {
    setSelected(child);

    if (onSelect) {
      onSelect(child);
    }
  };

  const renderRadio = isSelected => {
    return (
      <View style={styles.radioOuter}>
        {isSelected && <View style={styles.radioInner} />}
      </View>
    );
  };

  return (
    <ScrollView contentContainerStyle={styles.container}>
      <Text style={styles.heading}>{title}</Text>

      {data.map(item => (
        <View key={item.id} style={styles.sectionContainer}>
          {/* Parent */}
          <TouchableOpacity
            style={styles.row}
            onPress={() => toggleExpand(item.title)}>
            {renderRadio(expanded === item.title)}

            <Text style={styles.parentText}>{item.title}</Text>
          </TouchableOpacity>

          {/* Children */}
          {expanded === item.title && (
            <View style={styles.childContainer}>
              {item.children.map((child, index) => (
                <TouchableOpacity
                  key={index}
                  style={styles.row}
                  onPress={() => handleSelect(child)}>
                  {renderRadio(selected === child)}

                  <Text style={styles.childText}>{child}</Text>
                </TouchableOpacity>
              ))}
            </View>
          )}
        </View>
      ))}
    </ScrollView>
  );
};

export default RadioButtonTree;

const styles = StyleSheet.create({
  container: {
    padding: 20,
    backgroundColor: '#fff',
  },
  heading: {
    fontSize: 22,
    fontWeight: '700',
    marginBottom: 20,
  },
  sectionContainer: {
    marginBottom: 20,
  },
  row: {
    flexDirection: 'row',
    alignItems: 'center',
    marginVertical: 8,
  },
  parentText: {
    fontSize: 18,
    fontWeight: '600',
    marginLeft: 12,
  },
  childContainer: {
    paddingLeft: 40,
    marginTop: 5,
  },
  childText: {
    fontSize: 16,
    marginLeft: 12,
  },
  radioOuter: {
    width: 24,
    height: 24,
    borderRadius: 12,
    borderWidth: 2,
    borderColor: '#555',
    justifyContent: 'center',
    alignItems: 'center',
  },
  radioInner: {
    width: 12,
    height: 12,
    borderRadius: 6,
    backgroundColor: '#2196F3',
  },
});
