import React, { useState } from 'react';
import {
  View,
  Text,
  TouchableOpacity,
  StyleSheet,
  ScrollView,
} from 'react-native';

const subjectsData = [
  {
    id: 1,
    title: 'Science',
    children: ['Biology', 'Physics', 'Chemistry'],
  },
  {
    id: 2,
    title: 'Math',
    children: ['Algebra', 'Geometry', 'Trigonometry'],
  },
];

const RadioButtonTree = () => {
  const [expanded, setExpanded] = useState('Science');
  const [selected, setSelected] = useState('Physics');

  const toggleExpand = title => {
    if (expanded === title) {
      setExpanded('');
    } else {
      setExpanded(title);
    }
  };

  const renderRadio = isSelected => {
    return (
      <View style={styles.radioOuter}>
        {isSelected && <View style={styles.radioInner} />}
      </View>
    );
  };

  return (
    <ScrollView contentContainerStyle={styles.container}>
      <Text style={styles.heading}>Subject</Text>

      {subjectsData.map(item => (
        <View key={item.id} style={styles.sectionContainer}>
          {/* Parent */}
          <TouchableOpacity
            style={styles.row}
            onPress={() => toggleExpand(item.title)}>
            {renderRadio(expanded === item.title)}

            <Text style={styles.parentText}>{item.title}</Text>
          </TouchableOpacity>

          {/* Children */}
          {expanded === item.title && (
            <View style={styles.childContainer}>
              {item.children.map((child, index) => (
                <TouchableOpacity
                  key={index}
                  style={styles.row}
                  onPress={() => setSelected(child)}>
                  {renderRadio(selected === child)}

                  <Text style={styles.childText}>{child}</Text>
                </TouchableOpacity>
              ))}
            </View>
          )}
        </View>
      ))}

      <View style={styles.selectedBox}>
        <Text style={styles.selectedText}>
          Selected Subject: {selected}
        </Text>
      </View>
    </ScrollView>
  );
};

export default RadioButtonTree;

const styles = StyleSheet.create({
  container: {
    flexGrow: 1,
    padding: 20,
    backgroundColor: '#fff',
  },

  heading: {
    fontSize: 22,
    fontWeight: '700',
    marginBottom: 20,
    color: '#000',
  },

  sectionContainer: {
    marginBottom: 20,
  },

  row: {
    flexDirection: 'row',
    alignItems: 'center',
    marginVertical: 8,
  },

  parentText: {
    fontSize: 18,
    fontWeight: '600',
    marginLeft: 12,
    color: '#000',
  },

  childContainer: {
    paddingLeft: 40,
    marginTop: 5,
  },

  childText: {
    fontSize: 16,
    marginLeft: 12,
    color: '#333',
  },

  radioOuter: {
    width: 24,
    height: 24,
    borderRadius: 12,
    borderWidth: 2,
    borderColor: '#555',
    justifyContent: 'center',
    alignItems: 'center',
  },

  radioInner: {
    width: 12,
    height: 12,
    borderRadius: 6,
    backgroundColor: '#2196F3',
  },

  selectedBox: {
    marginTop: 30,
    padding: 15,
    backgroundColor: '#F1F5F9',
    borderRadius: 10,
  },

  selectedText: {
    fontSize: 16,
    fontWeight: '600',
    color: '#000',
  },
});
